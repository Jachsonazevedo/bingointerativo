    <script>
        var $ = function(id) { return document.getElementById(id); };
        var audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        
        // Bots agora com Emojis
        var botNames = [
            "🤖 Maria", "🤠 José", "👩 Ana", "👴 Carlos", 
            "🧑‍🦱 Pedro", "👵 Lúcia", "👨‍✈️ Paulo", "👩‍🏫 Fernanda", 
            "👮 Roberto", "👩‍⚕️ Carla", "👸 Dona Iza", "🤴 Sr. João", "🧚 Bete"
        ];
        
        // Frases focadas em benefícios
        var phrases = [
            "Exercita a memória!", 
            "Melhora o foco!", 
            "Estimula o raciocínio!",
            "Mantém a mente ativa!", 
            "Diversão saudável!", 
            "Sua mente agradece!",
            "Atenção na cartela!", 
            "Respire e divirta-se!"
        ];
        
        var S = {
            scr: 'welcome', name: '', bal: 500, mute: false,
            cfg: { cards: 1, rule: 'line', bet: 10, spd: 4500, botCount: 3 },
            game: { active: false, pause: true, drawn: [], timer: null, cards: [], marks: [], bots: [] }
        };

        // --- AUDIO ---
        function resumeAudio() { if (audioCtx.state === 'suspended') audioCtx.resume(); }

        function playSound(type) {
            if(S.mute) return;
            resumeAudio();
            var now = audioCtx.currentTime;
            var osc = audioCtx.createOscillator();
            var gain = audioCtx.createGain();
            osc.connect(gain); gain.connect(audioCtx.destination);

            if(type === 'click') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(800, now);
                gain.gain.setValueAtTime(0.1, now); osc.start(); osc.stop(now+0.1);
            } else if(type === 'draw') {
                osc.type = 'triangle'; osc.frequency.setValueAtTime(500, now);
                osc.frequency.linearRampToValueAtTime(1000, now+0.1);
                gain.gain.setValueAtTime(0.2, now); gain.gain.exponentialRampToValueAtTime(0.01, now+0.5);
                osc.start(); osc.stop(now+0.5);
            } else if(type === 'win') {
                [523, 659, 783, 1046].forEach(function(f, i) {
                    var o = audioCtx.createOscillator(); var g = audioCtx.createGain();
                    o.connect(g); g.connect(audioCtx.destination);
                    o.type = 'square'; o.frequency.value = f;
                    g.gain.setValueAtTime(0.05, now + i*0.1); g.gain.linearRampToValueAtTime(0, now + i*0.1 + 0.4);
                    o.start(now + i*0.1); o.stop(now + i*0.1 + 0.4);
                });
            } else if(type === 'lose') {
                osc.type = 'sawtooth'; osc.frequency.setValueAtTime(150, now);
                osc.frequency.linearRampToValueAtTime(100, now+0.5);
                gain.gain.setValueAtTime(0.2, now); gain.gain.linearRampToValueAtTime(0, now+0.5);
                osc.start(); osc.stop(now+0.5);
            } else if(type === 'alert') {
                osc.type = 'square'; osc.frequency.setValueAtTime(300, now);
                gain.gain.setValueAtTime(0.1, now); 
                osc.frequency.linearRampToValueAtTime(200, now+0.3);
                osc.start(); osc.stop(now+0.3);
            }
        }

        function speak(text) {
            if(S.mute || !('speechSynthesis' in window)) return;
            window.speechSynthesis.cancel();
            var u = new SpeechSynthesisUtterance(text);
            u.lang = 'pt-BR'; u.rate = 1.0;
            window.speechSynthesis.speak(u);
        }

        // --- INIT ---
        window.onload = function() {
            var sn = localStorage.getItem('bingo_nm');
            var sb = localStorage.getItem('bingo_bl');
            if(sb) S.bal = parseInt(sb);
            if(sn) { S.name = sn; $('inp-name').value = sn; }
            updUi();
        };

        function updUi() {
            $('ui-bal').innerText = S.bal;
            localStorage.setItem('bingo_bl', S.bal);
        }

        function switchScr(id) {
            document.querySelectorAll('section').forEach(function(el) {
                el.classList.add('hidden'); el.classList.remove('flex');
            });
            $(id).classList.remove('hidden'); $(id).classList.add('flex');
        }

        // --- CONFIG ---
        function goConfig() {
            resumeAudio();
            var n = $('inp-name').value.trim();
            if(!n) return alert('Por favor, digite seu nome.');
            S.name = n; localStorage.setItem('bingo_nm', n);
            switchScr('screen-config');
            playSound('click');
            setCards(1); setBet(10); setBots(3);
        }

        function setCards(n) {
            S.cfg.cards = n;
            $('btn-c1').classList.toggle('active', n===1);
            $('btn-c2').classList.toggle('active', n===2);
            playSound('click');
        }

        function setBots(n) {
            S.cfg.botCount = n;
            [1,3,7,10].forEach(k => {
                var el = $('btn-b'+k);
                if(el) el.classList.toggle('active', k===n);
            });
            playSound('click');
        }

        function setBet(n) {
            if(n > S.bal) { playSound('lose'); return alert('Saldo insuficiente!'); }
            S.cfg.bet = n; updatePot(); playSound('click');
        }
        function updatePot() {
            var m = $('sel-rule').value === 'full' ? 5 : 2;
            $('ui-pot').innerText = S.cfg.bet * m;
        }

        function toggleMute() {
            S.mute = !S.mute;
            $('icon-sound').className = S.mute ? 'fas fa-volume-mute text-red-500 text-xs' : 'fas fa-volume-up text-xs';
            if(S.mute) window.speechSynthesis.cancel();
        }

        // --- GAME LOGIC ---
        function startGame() {
            resumeAudio();
            if(S.bal < S.cfg.bet) return alert('Sem saldo.');
            S.bal -= S.cfg.bet; updUi();
            
            S.game = { active: true, pause: true, drawn: [], timer: null, cards: [], marks: [], bots: [] };
            
            // Setup UI
            $('ball-history').innerHTML = ''; 
            
            // Add Emoji to Player Name
            $('game-pname').innerText = "😎 " + S.name;
            
            $('game-bal').innerText = S.bal;
            var m = $('sel-rule').value === 'full' ? 5 : 2;
            $('game-pot').innerText = (S.cfg.bet * m);
            
            // Rule Headline
            var ruleText = $('sel-rule').value === 'full' ? "COMPLETAR A CARTELA" : "FAZER UMA LINHA";
            $('game-rule-text').innerText = "NESTA RODADA GANHA QUEM: " + ruleText;

            var btn = $('btn-play');
            btn.innerHTML = '<i class="fas fa-play"></i> <span>COMEÇAR</span>';
            btn.className = "flex-grow h-10 rounded-lg btn-action text-sm flex items-center justify-center gap-2 shadow-lg uppercase tracking-wide";
            
            $('game-msg').innerText = "Divirta-se e exercite a mente!";
            $('game-msg').className = "text-blue-100 text-[10px] uppercase tracking-widest font-bold";

            // Setup Bots
            var needed = $('sel-rule').value === 'full' ? 24 : 5;
            var shuffled = botNames.sort(() => 0.5 - Math.random());
            for(var i=0; i<S.cfg.botCount; i++) {
                var bName = shuffled[i] || ("🤖 Bot " + (i+1));
                S.game.bots.push({ name: bName, needed: needed });
            }
            updateRivalsUI();

            genCards();
            renderCards();
            switchScr('screen-game');
            playSound('click');
        }

        function updateRivalsUI() {
            var html = "";
            var alertMsg = "";
            
            S.game.bots.forEach(b => {
                var styleClass = "";
                if (b.needed === 3) styleClass = "warn-3";
                if (b.needed === 2) styleClass = "warn-2";
                if (b.needed === 1) styleClass = "warn-1";

                html += `<div class="rival-pill ${styleClass}">
                    <span class="text-white font-bold">${b.name}</span>
                    <span class="text-blue-200 text-[9px] ml-1">falta ${b.needed}</span>
                </div>`;
                
                // Priority Alerts
                if(b.needed === 1) alertMsg = "⚠️ " + b.name + " ESTÁ POR UMA!";
                else if(!alertMsg && b.needed === 2) alertMsg = "Atenção: " + b.name + " está por duas!";
            });
            $('rivals-bar').innerHTML = html;
            
            // Update Main Alert Message if critical
            if(alertMsg) {
                $('game-msg').innerText = alertMsg;
                $('game-msg').className = "text-yellow-300 text-xs uppercase tracking-widest font-bold alert-text";
                if(alertMsg.includes("POR UMA")) playSound('alert');
            }
        }

        function genCards() {
            var ranges = [[1,15],[16,30],[31,45],[46,60],[61,75]];
            for(var c=0; c<S.cfg.cards; c++){
                var flat = new Array(25).fill(0);
                for(var col=0; col<5; col++){
                    var nums = [];
                    while(nums.length<5){
                        var r = Math.floor(Math.random()*(ranges[col][1]-ranges[col][0]+1))+ranges[col][0];
                        if(!nums.includes(r)) nums.push(r);
                    }
                    for(var row=0; row<5; row++) flat[row*5+col] = nums[row];
                }
                flat[12] = 0; // Free
                S.game.cards.push(flat);
                var m = new Array(25).fill(false); m[12] = true;
                S.game.marks.push(m);
            }
        }

        function renderCards() {
            var area = $('game-area'); area.innerHTML = '';
            var wrap = document.createElement('div');
            wrap.className = "cards-wrapper";

            S.game.cards.forEach(function(card, cIdx) {
                var div = document.createElement('div');
                div.className = "bingo-card";
                
                ['B','I','N','G','O'].forEach(function(l) {
                    var h = document.createElement('div');
                    h.className = "header-cell"; h.innerText = l;
                    div.appendChild(h);
                });

                card.forEach(function(n, i) {
                    var cell = document.createElement('div');
                    cell.className = "num-cell"; cell.id = `c${cIdx}-${i}`;
                    
                    if(n === 0) {
                        cell.classList.add('marked', 'free');
                        cell.innerHTML = '<i class="fas fa-star text-yellow-600"></i>';
                    } else {
                        cell.innerText = n;
                        cell.onclick = function() { hit(cIdx, i, n); };
                    }
                    div.appendChild(cell);
                });
                wrap.appendChild(div);
            });
            area.appendChild(wrap);
        }

        // --- GAMEPLAY ---
        function togglePlay() {
            if(S.game.pause) {
                S.game.pause = false;
                var btn = $('btn-play');
                btn.innerHTML = '<i class="fas fa-pause"></i> <span>PAUSAR</span>';
                btn.className = "flex-grow h-10 rounded-lg btn-yellow text-sm flex items-center justify-center gap-2 shadow-lg uppercase tracking-wide";
                
                if(S.game.drawn.length === 0) draw();
                S.game.timer = setInterval(draw, S.cfg.spd);
            } else { 
                stop(); 
                var btn = $('btn-play');
                btn.innerHTML = '<i class="fas fa-play"></i> <span>CONTINUAR</span>';
                btn.className = "flex-grow h-10 rounded-lg btn-action text-sm flex items-center justify-center gap-2 shadow-lg uppercase tracking-wide";
            }
        }

        function stop() {
            S.game.pause = true; clearInterval(S.game.timer);
        }

        function draw() {
            if(!S.game.active || (S.game.pause && S.game.timer)) return;
            if(S.game.drawn.length >= 75) { stop(); alert("Fim das pedras!"); return; }

            // Motivational Phrase
            if(Math.random() < 0.2 && !$('game-msg').innerText.includes("⚠️")) {
                $('game-msg').innerText = phrases[Math.floor(Math.random()*phrases.length)];
                $('game-msg').className = "text-blue-100 text-[10px] uppercase tracking-widest font-bold";
            }

            setTimeout(function() {
                if(!S.game.active) return;
                var n; do { n = Math.floor(Math.random()*75)+1; } while(S.game.drawn.includes(n));
                S.game.drawn.push(n);

                // Update History
                var hist = $('ball-history');
                if(S.game.drawn.length === 1) hist.innerHTML = '';
                var prev = hist.querySelector('.latest');
                if(prev) prev.classList.remove('latest');

                var ballEl = document.createElement('div');
                ballEl.className = "ball-history-item latest";
                ballEl.innerText = n;
                ballEl.animate([{ transform: 'scale(0)' }, { transform: 'scale(1)' }], { duration: 300 });

                hist.prepend(ballEl);
                
                playSound('draw');
                speak(n.toString());

                // Bot Logic
                S.game.bots.forEach(b => {
                    if(b.needed > 0 && Math.random() < 0.22) { 
                        b.needed--;
                    }
                });
                updateRivalsUI();
                
                // Check Bot Win
                var winnerBot = S.game.bots.find(b => b.needed === 0);
                if(winnerBot) {
                    doEnd(false, winnerBot.name);
                }

            }, 500);
        }

        function hit(cIdx, i, n) {
            if(S.game.marks[cIdx][i]) return;
            var el = $(`c${cIdx}-${i}`);
            
            if(S.game.drawn.includes(n)) {
                S.game.marks[cIdx][i] = true;
                el.classList.add('marked');
                playSound('click');
                checkWin(cIdx);
            } else {
                el.classList.add('shake');
                setTimeout(() => el.classList.remove('shake'), 400);
                playSound('lose');
            }
        }

        function checkWin(cIdx) {
            var m = S.game.marks[cIdx];
            var r = $('sel-rule').value;
            var w = false;
            var _chk = function(arr) { return arr.every(function(x) { return m[x]; }); };

            if(r === 'line') {
                for(var k=0; k<5; k++) {
                    if(_chk([k*5, k*5+1, k*5+2, k*5+3, k*5+4])) w=true; 
                    if(_chk([k, k+5, k+10, k+15, k+20])) w=true;
                }
                if(_chk([0,6,12,18,24]) || _chk([4,8,12,16,20])) w=true;
            } else {
                if(m.every(function(x) { return x; })) w=true;
            }

            if(w) doEnd(true, S.name);
        }

        function doEnd(won, winnerName) {
            stop(); S.game.active = false;
            var border = $('modal-border');
            var title = $('end-title');
            var msg = $('end-msg');
            var pbox = $('prize-box');
            
            if(won) {
                var mult = $('sel-rule').value === 'full' ? 5 : 2;
                var prize = S.cfg.bet * mult;
                S.bal += prize; updUi();
                
                border.className = "bg-blue-900/90 border-4 border-yellow-500 rounded-3xl p-6 text-center max-w-xs w-full relative overflow-hidden shadow-2xl backdrop-blur-xl";
                title.innerText = "BINGO!";
                title.className = "text-5xl text-yellow-400 vegas-font mb-2";
                msg.innerText = "Parabéns, " + S.name + "!";
                pbox.style.display = "block";
                $('win-amount').innerText = prize;
                playSound('win');
            } else {
                border.className = "bg-red-900/90 border-4 border-red-500 rounded-3xl p-6 text-center max-w-xs w-full relative overflow-hidden shadow-2xl backdrop-blur-xl";
                title.innerText = "DERROTA";
                title.className = "text-4xl text-red-500 vegas-font mb-2";
                msg.innerText = winnerName + " bateu o Bingo!";
                pbox.style.display = "none";
                playSound('lose');
            }
            
            $('modal-end').classList.remove('hidden');
        }

        function closeEnd() {
            $('modal-end').classList.add('hidden');
            switchScr('screen-config');
        }

        function quit() {
            if(confirm('Deseja realmente sair?')) {
                stop(); window.speechSynthesis.cancel();
                switchScr('screen-config');
            }
        }
